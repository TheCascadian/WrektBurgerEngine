# Spheres of influence and leagues (great_powers feature).
build_influence = {
	name = "Build Influence"
	desc = "Spend our diplomats on this country: influence grows 1 a month, and above 50 it can be drawn into our sphere."
	envoy = no
	potential = {
		has_feature = great_powers
		is_great_power = yes
		NOT = { is_campaigning_influence = scope:recipient }
		NOT = { in_sphere_of = ROOT }
	}
	allow = { can_campaign_influence = scope:recipient }
	on_accept = { start_influence = scope:recipient }
	ai_will_do = { base = 0 }
}

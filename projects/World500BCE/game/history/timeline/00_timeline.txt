# gets the snapshot nearest to it. This file holds the tabs and the dates of the start screen, and any events that should happen between two snapshots:
#   era = { name = "..." from = Y.M.D to = Y.M.D desc = "..." suggested = { Y.M.D ... } }   a tab of the start screen; suggested dates should be bookmarks
#   bookmark = { date = Y.M.D name = "..." desc = "..." }                                   a date to pick (make it the date of a snapshot)
#   event = { date = Y.M.D name = "..." <operations> }                                      applied after the snapshot before it
#     transfer = { from = TAG to = TAG regions = { a b } areas = { x } count = N }   what 'from' owns; regions (state key without its number), areas (areas*.txt) and count are optional
#     give     = { to = TAG regions = { a } areas = { x } count = N [from = TAG] }   the states of these regions or areas, whoever owns them
#     settle   = { to = TAG regions = { a } areas = { x } count = N }                states nobody owns
#     annex    = { from = TAG to = TAG }                                              every state, army and fleet of 'from'
#     rename   = { tag = TAG name = "..." }

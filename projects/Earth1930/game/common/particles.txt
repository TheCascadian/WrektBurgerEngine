# Particle effects. One block per effect; any figure clip event, the console (`particle name`) or a script (`spawn_particles = name`) can show one by name.
# Sizes, speeds and gravity are in metres at the scale of a soldier; the game scales them up with the figures.
#
#   name = {
#       count     = 12                       # particles per burst
#       life      = { 0.4 0.9 }              # seconds, min and max (one number: fixed)
#       speed     = { 0.5 2.0 }              # metres per second
#       size      = { 0.10 0.30 }            # width at birth and at death
#       color     = { FFF4C0FF FF8020A0 00000000 }   # RRGGBB or RRGGBBAA, blended over the particle's life
#       gravity   = 0                        # positive falls, negative rises
#       drag      = 1.5                      # 0 = none, higher slows faster
#       spread    = 25                       # degrees: half-angle of the cone about `direction`
#       direction = forward                  # forward (the way a muzzle points) | up | out (all ways)
#       additive  = yes                      # yes for light (flashes, sparks), no for smoke and dust
#   }
#
# The engine's own effects (muzzle_flash, muzzle_smoke, dust_puff) are the defaults; define one with the same name to replace it.
